68 template<class VY>
69 ExecStatus
70 BoolBase<VY>::advise(Space& home, Advisor& _a, const Delta&) {
71 ViewAdvisor<BoolView>& a(static_cast<ViewAdvisor<BoolView>&>(_a));
72 ExecStatus es;
73 if (status == (VS_ZERO | VS_ONE)) {
74 // Everything is already decided
75 es = ES_FIX;
76 } else {
77 if (a.view().zero())
78 status |= VS_ZERO;
79 else
80 status |= VS_ONE;
81 es = ES_NOFIX;
82 }
83 a.dispose(home,c);
84 if (c.empty())
85 es = ES_NOFIX;
86 return es;
87 }
