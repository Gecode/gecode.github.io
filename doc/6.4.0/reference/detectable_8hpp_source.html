38 detectable(Space& home, TaskViewArray<ManTaskView>& t) {
39 sort<ManTaskView,STO_ECT,true>(t);
40
41 Region r;
42
43 OmegaTree<ManTaskView> o(r,t);
44 TaskViewIter<ManTaskView,STO_LST,true> q(r,t);
45 int* est = r.alloc<int>(t.size());
46

70 template<class OptTaskView, class PL>
71 forceinline ExecStatus
72 detectable(Space& home, Propagator& p, TaskViewArray<OptTaskView>& t) {
73 sort<OptTaskView,STO_ECT,true>(t);
74
75 Region r;
76
77 OmegaTree<OptTaskView> o(r,t);
78 ManTaskViewIter<OptTaskView,STO_LST,true> q(r,t);
79 int* est = r.alloc<int>(t.size());
80
81 for (int i=0; i<t.size(); i++) {
82 while (q() && (t[i].ect() > t[q.task()].lst())) {
83 o.insert(q.task()); ++q;
84 }
85 est[i] = o.ect(i);
86 }
87
88 int n = t.size();
89 for (int i=n; i--; )
90 if (t[i].mandatory()) {
91 GECODE_ME_CHECK(t[i].est(home,est[i]));
92 } else if (est[i] > t[i].lst()) {
93 GECODE_ME_CHECK(t[i].excluded(home));
94 t[i].cancel(home,p,PL::pc); t[i]=t[--n];
95 }
96 t.size(n);
97
98 return (t.size() < 2) ? home.ES_SUBSUMED(p) : ES_OK;
99 }
