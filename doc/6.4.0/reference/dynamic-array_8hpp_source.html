104 template<class T, class A>
105 forceinline const DynamicArray<T,A>&
106 DynamicArray<T,A>::operator =(const DynamicArray<T,A>& da) {
107 if (this != &da) {
108 if (n < da.n) {
109 a.free(x,n); n = da.n; x = a.template alloc<T>(n);
110 }
111 (void) heap.copy(x,da.x,n);
112 }
113 return *this;
114 }

116 template<class T, class A>
117 void
118 DynamicArray<T,A>::resize(int i) {
119 int m = std::max(i+1, (3*n)/2);
120 x = a.realloc(x,n,m);
121 n = m;
122 }
