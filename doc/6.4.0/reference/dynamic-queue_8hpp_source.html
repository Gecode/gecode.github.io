92 template<class T, class A>
93 void
94 DynamicQueue<T,A>::resize(void) {
95 assert(fst == lst);
96 T* nq = a.template alloc<T>(limit << 1);
97 int j=0;
98 for (int i = fst; i<limit; i++)
99 nq[j++] = q[i];
100 for (int i = 0; i<lst; i++)
101 nq[j++] = q[i];
102 a.template free<T>(q,limit);
103 q = nq;
104 fst = 0;
105 lst = limit;
106 limit <<= 1;
107 }
