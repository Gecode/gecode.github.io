43 template<class Char, class Traits, class View>
44 std::basic_ostream<Char,Traits>&
45 print_view(std::basic_ostream<Char,Traits>& os, const View& x) {
46 std::basic_ostringstream<Char,Traits> s;
47 s.copyfmt(os); s.width(0);
48 if (x.assigned()) {
49 s << "[[" << x.med() << "]]";
50 } else {
51 s << "[[" << x.min() << ".." << x.max() << "]]";
52 }
53 return os << s.str();
54 }
