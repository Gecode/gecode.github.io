146 forceinline IntType
147 u_type(unsigned int n) {
148 if (n < UCHAR_MAX)
149 return IT_CHAR;
150 else if (n < USHRT_MAX)
151 return IT_SHRT;
152 else
153 return IT_INT;
154 }
155
156 forceinline IntType
157 s_type(int n) {
158 if ((n > SCHAR_MIN) && (n < SCHAR_MAX))
159 return IT_CHAR;
160 else if ((n > SHRT_MIN) && (n < SHRT_MAX))
161 return IT_SHRT;
162 else
163 return IT_INT;
164 }
