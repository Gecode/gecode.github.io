36 template<class IntType>
37 forceinline IntType
38 ceil_div_pp(IntType x, IntType y) {
39 assert((x >= 0) && (y >= 0));
40 /*
41 * This is smarter than it looks: for many cpus, % and / are
42 * implemented by the same instruction and an optimizing compiler
43 * will generate the instruction only once.
44 */
45 return ((x % y) == 0) ? x/y : (x/y + 1);
46 }
47 template<class IntType>
48 forceinline IntType
49 floor_div_pp(IntType x, IntType y) {
50 assert((x >= 0) && (y >= 0));
51 return x / y;
52 }

54 template<class IntType>
55 forceinline IntType
56 ceil_div_px(IntType x, IntType y) {
57 assert(x >= 0);
58 return (y >= 0) ? ceil_div_pp(x,y) : -floor_div_pp(x,-y);
59 }
60 template<class IntType>
61 forceinline IntType
62 floor_div_px(IntType x, IntType y) {
63 assert(x >= 0);
64 return (y >= 0) ? floor_div_pp(x,y) : -ceil_div_pp(x,-y);
65 }

69 ceil_div_xp(IntType x, IntType y) {
