44 template<class Box>
45 forceinline int
46 Base<Box>::partition(Box* b, int i, int n) {
47 int j = n-1;
48 while (true) {
49 while (!b[j].mandatory() && (--j >= 0)) {}
50 while (b[i].mandatory() && (++i < n)) {}
51 if (j <= i) break;
52 std::swap(b[i],b[j]);
53 }
54 return i;
55 }
