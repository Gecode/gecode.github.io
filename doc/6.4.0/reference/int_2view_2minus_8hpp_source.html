296 inline int
297 MinusView::med(void) const {
298 if (x.range())
299 return (min()+max())/2 - ((min()+max())%2 < 0 ? 1 : 0);
300
301 unsigned int i = x.size() / 2;
302 if (size() % 2 == 0)
303 i--;
304 ViewRanges<MinusView> r(*this);
305 while (i >= r.width()) {
306 i -= r.width();
307 ++r;
308 }
309 return r.min() + static_cast<int>(i);
310 }
