38 template<class Char, class Traits, class View>
39 std::basic_ostream<Char,Traits>&
40 print_view(std::basic_ostream<Char,Traits>& os, const View& x) {
41 std::basic_ostringstream<Char,Traits> s;
42 s.copyfmt(os); s.width(0);
43 if (x.assigned()) {
44 s << x.val();
45 } else if (x.range()) {
46 s << '[' << x.min() << ".." << x.max() << ']';
47 } else {
48 s << '{';
49 ViewRanges<View> r(x);
50 while (true) {
51 if (r.min() == r.max()) {
52 s << r.min();
53 } else {
54 s << r.min() << ".." << r.max();
55 }
56 ++r;
57 if (!r()) break;
58 s << ',';
59 }
60 s << '}';
61 }
62 return os << s.str();
63 }
