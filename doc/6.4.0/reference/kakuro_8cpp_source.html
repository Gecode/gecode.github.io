400 TupleSet generate(int n, int c) {
401 // Setup search engine that enumerates all solutions
402 DistinctLinear* e = new DistinctLinear(n,c);
403 DFS<DistinctLinear> d(e);
404 delete e;
405 TupleSet ts(n);
406 while (DistinctLinear* s = d.next()) {
407 ts.add(s->solution()); delete s;

414 class Cache {
415 private:
417 class Entry {
418 public:
419 int n;
420 int c;
421 TupleSet ts;
422 Entry* next;
423 };
424 Entry* cache;
425 public:
427 Cache(void) : cache(nullptr) {}
429 TupleSet get(int n, int c) {
430 for (Entry* e = cache; e != nullptr; e = e->next)
431 if ((e->n == n) && (e->c == c))
432 return e->ts;
433 {
434 Entry* e = new Entry;
435 e->n = n;
436 e->c = c;
437 e->ts = generate(n,c);
438 e->next = cache;
439 cache = e;
440 }
441 return cache->ts;
442 }
444 ~Cache(void) {
445 Entry* e = cache;
446 while (e != nullptr) {
447 Entry* d = e;
448 e = e->next;
449 delete d;
450 }
451 }
452 };

467class Kakuro : public Script {
468protected:
469 const int w;
470 const int h;
471 IntVarArray f;
472public:
474 enum {
475 MODEL_DECOMPOSE,
476 MODEL_COMBINE,
477 };
479 IntVar init(IntVar& x) {
480 if (x.min() == 0)
481 x = IntVar(*this,1,9);
482 return x;
483 }
484
485 void distinctlinear(Cache& dc, const IntVarArgs& x, int c,
486 const SizeOptions& opt) {
487 int n=x.size();
488 if (opt.model() == MODEL_DECOMPOSE) {
489 if (n < 8)
490 linear(*this, x, IRT_EQ, c, opt.ipl());
491 else if (n == 8)
492 rel(*this, x, IRT_NQ, 9*(9+1)/2 - c);
493 distinct(*this, x, opt.ipl());
494 } else {
495 switch (n) {
496 case 0:
497 return;
498 case 1:
499 rel(*this, x[0], IRT_EQ, c);
500 return;
501 case 8:
502 // Prune the single missing digit
503 rel(*this, x, IRT_NQ, 9*(9+1)/2 - c);
504 break;
505 case 9:
506 break;
507 default:
508 if (c == n*(n+1)/2) {
509 // sum has unique decomposition: 1 + ... + n
510 rel(*this, x, IRT_LQ, n);
511 } else if (c == n*(n+1)/2 + 1) {
512 // sum has unique decomposition: 1 + ... + n-1 + n+1
513 rel(*this, x, IRT_LQ, n+1);
514 rel(*this, x, IRT_NQ, n);
515 } else if (c == 9*(9+1)/2 - (9-n)*(9-n+1)/2) {
516 // sum has unique decomposition: (9-n+1) + (9-n+2) + ... + 9
517 rel(*this, x, IRT_GQ, 9-n+1);
518 } else if (c == 9*(9+1)/2 - (9-n)*(9-n+1)/2 + 1) {
519 // sum has unique decomposition: (9-n) + (9-n+2) + ... + 9
520 rel(*this, x, IRT_GQ, 9-n);
521 rel(*this, x, IRT_NQ, 9-n+1);
522 } else {
523 extensional(*this, x, dc.get(n,c));
524 return;
525 }
526 }
527 distinct(*this, x, opt.ipl());
528 }
529 }
530
531 Kakuro(const SizeOptions& opt)
532 : Script(opt),
533 w(examples[opt.size()][0]), h(examples[opt.size()][1]),
534 f(*this,w*h) {
535 IntVar black(*this,0,0);
536 // Initialize all fields as black (unused). Only if a field
537 // is actually used in a constraint, create a fresh variable
538 // for it (done via init).
539 for (int i=w*h; i--; )
540 f[i] = black;
541
542 // Cache of already computed tuple sets
543 Cache cache;
544
545 // Matrix for accessing board fields
546 Matrix<IntVarArray> b(f,w,h);
547 // Access to hints
548 const int* k = &examples[opt.size()][2];
549
550 // Process vertical hints
551 while (*k >= 0) {
552 int x=*k++; int y=*k++; int n=*k++; int s=*k++;
553 IntVarArgs col(n);
554 for (int i=n; i--; )
555 col[i]=init(b(x,y+i+1));
556 distinctlinear(cache,col,s,opt);
557 }
558 k++;
559
560 // Process horizontal hints
561 while (*k >= 0) {
562 int x=*k++; int y=*k++; int n=*k++; int s=*k++;
563 IntVarArgs row(n);
564 for (int i=n; i--; )
565 row[i]=init(b(x+i+1,y));
566 distinctlinear(cache,row,s,opt);
567 }
568 branch(*this, f, INT_VAR_AFC_SIZE_MAX(opt.decay()), INT_VAL_SPLIT_MIN());
569 }

571 Kakuro(Kakuro& s) : Script(s), w(s.w), h(s.h) {
572 f.update(*this, s.f);
573 }

575 virtual Space*
576 copy(void) {
577 return new Kakuro(*this);
578 }

580 virtual void
581 print(std::ostream& os) const {
582 Matrix<IntVarArray> b(f,w,h);
583 for (int y=0; y<h; y++) {
584 os << '\t';
585 for (int x=0; x<w; x++)
586 if (b(x,y).min() == 0)
587 os << ". ";
588 else
589 os << b(x,y) << ' ';
590 os << std::endl;
591 }
592 }
593};
