518 forceinline void
519 Space::recover(Space& source) {
520 ActorLink* clone_lists[2] = {&pl, &bl};
521 for (int i=0; i<2; i++) {
522 ActorLink* l = clone_lists[i];
523 ActorLink* c = l->next();
524 while (c != l) {
525 ActorLink* n = c->next();
526 (void) Actor::cast(c)->dispose(*this);
527 c = n;
528 }
529 l->init();
530 }
531 while (pc.c.local != nullptr) {
532 LocalObject* l = pc.c.local;
533 ActorLink* n = l->next();
534 pc.c.local = (n == nullptr) ? nullptr : LocalObject::cast(n);
535 ActorLink* fwd = l->prev();
536 if (fwd != nullptr) {
537 (void) Actor::cast(fwd)->dispose(*this);
538 l->prev(nullptr);
539 }
540 }
541
542 ActorLink* lists[2] = {&source.pl, &source.bl};
543 for (int i=0; i<2; i++) {
544 ActorLink* l = lists[i];
545 ActorLink* p_a = l;
546 ActorLink* c_a = p_a->next();
547 while (c_a != l) {
548 if (i == 0) {
549 Propagator* p = Propagator::cast(c_a);
550 if (p->u.advisors != nullptr) {
551 ActorLink* a = p->u.advisors;
552 p->u.advisors = nullptr;
553 do {
554 a->prev(p); a = a->next();
555 } while (a != nullptr);
556 }
557 }
558 ActorLink* fwd = c_a->prev();
559 if (fwd != p_a) {
560 c_a->prev(p_a);
561 }
562 p_a = c_a; c_a = c_a->next();
563 }
564 }
565
566 ActorLink** sub = static_cast<ActorLink**>(mm.subscriptions());
567 recover_noidx();
568#ifdef GECODE_HAS_INT_VARS
569 Gecode::VarImp<Gecode::Int::IntVarImpConf>::recover(*this,sub);
570#endif
571#ifdef GECODE_HAS_INT_VARS
572 Gecode::VarImp<Gecode::Int::BoolVarImpConf>::recover(*this,sub);
573#endif
574#ifdef GECODE_HAS_SET_VARS
575 Gecode::VarImp<Gecode::Set::SetVarImpConf>::recover(*this,sub);
576#endif
577#ifdef GECODE_HAS_FLOAT_VARS
578 Gecode::VarImp<Gecode::Float::FloatVarImpConf>::recover(*this,sub);
579#endif
