36 template<class OptTask, class PL>
37 ExecStatus
38 purge(Space& home, Propagator& p, TaskArray<OptTask>& t) {
39 int n=t.size();
40 for (int i=n; i--; )
41 if (t[i].excluded()) {
42 t[i].cancel(home,p,PL::pc); t[i]=t[--n];
43 }
44 t.size(n);
45
46 return (t.size() < 2) ? home.ES_SUBSUMED(p) : ES_OK;
47 }

49 template<class OptTask, class PL, class Cap>
50 ExecStatus
51 purge(Space& home, Propagator& p, TaskArray<OptTask>& t, Cap c) {
52 int n=t.size();
53 for (int i=n; i--; )
54 if (t[i].excluded()) {
55 t[i].cancel(home,p,PL::pc); t[i]=t[--n];
56 }
57 t.size(n);
58 if (t.size() == 1) {
59 if (t[0].mandatory())
60 GECODE_ME_CHECK(c.gq(home, t[0].c()));
61 else if (c.min() < t[0].c())
62 return ES_OK;
63 }
64
65 return (t.size() < 2) ? home.ES_SUBSUMED(p) : ES_OK;
66 }
