113 template<class I, class J>
114 inline void
115 Append<I,J>::operator ++(void) {
116 if (i()) {
117 mi = i.min(); ma = i.max();
118 ++i;
119 if (!i() && j() && (j.min() == ma+1)) {
120 ma = j.max();
121 ++j;
122 }
123 } else if (j()) {
124 mi = j.min(); ma = j.max();
125 ++j;
126 } else {
127 finish();
128 }
129 }

162 template<class I>
163 inline void
164 NaryAppend<I>::operator ++(void) {
165 mi = r[active].min();
166 ma = r[active].max();
167 ++r[active];
168 while (!r[active]()) {
169 //Skip empty iterators:
170 do {
171 active++;
172 if (active >= n) {
173 finish(); return;
174 }
175 } while (!r[active]());
176 if (r[active].min() == ma+1){
177 ma = r[active].max();
178 ++r[active];
179 } else {
180 return;
181 }
182 }
183 }
