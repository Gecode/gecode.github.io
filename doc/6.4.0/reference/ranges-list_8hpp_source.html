150 if (&i != this) {
151 if ((rlio != nullptr) && (--rlio->use_cnt == 0)) {
152 Region& r = rlio->allocator();
153 rlio->~RLIO();
154 r.rfree(rlio,sizeof(RLIO));

156 rlio = i.rlio;

158 rlio->use_cnt++;
159 c=i.c; h=i.h;
