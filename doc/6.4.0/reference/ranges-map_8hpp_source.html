161 Map<I,M,false>::next(void) {
162 if (i()) {
163 mi = m.min(i.min());
164 ma = m.max(i.max());
165 ++i;
166 while (i() && (ma+1 >= m.min(i.min()))) {
167 ma = m.max(i.max()); ++i;
168 }
169 } else {
170 finish();
171 }
