54 forceinline
55 RefCount::RefCount(unsigned long int n0) {
56 n.store(n0,std::memory_order_release);
57 }
58 forceinline void
59 RefCount::inc(void) {
60 (void) n.fetch_add(1U,std::memory_order_relaxed);
61 }
62 forceinline bool
63 RefCount::dec(void) {
64 return n.fetch_sub(1U,std::memory_order_acq_rel) == 1U;
65 }
66 forceinline
67 RefCount::operator bool(void) const {
68 return n.load(std::memory_order_acquire) > 0U;
69 }
