43 template<class View, class A>
44 inline
45 SupportValues<View,A>::SupportValues(A& a0, View x0)
46 : a(a0), x(x0), bs(a,x.size(),true) {
47 unsigned int n = 0;
48 for (ViewRanges<View> r(x); r(); ++r)
49 n++;
50 rp_fst = a.template alloc<RangePos>(n+1);
51 rp_lst = rp_fst + n;
52 unsigned int p = 0;
53 int i = 0;
54 for (ViewRanges<View> r(x); r(); ++r) {
55 rp_fst[i].min = r.min();
56 rp_fst[i].pos = p;
57 p += r.width(); i++;
58 }
59 rp_fst[i].pos=p;
60 reset();
61 }

98 template<class View, class A>
99 forceinline bool
100 SupportValues<View,A>::_support(int n) {
101 RangePos* l = rp_fst;
102 RangePos* r = rp_lst-1;
103 while (true) {
104 if (l > r) return false;
105 RangePos* m = l + (r-l)/2;
106 int max = m->min + static_cast<int>((m+1)->pos - m->pos) - 1;
107 if ((n >= m->min) && (n <= max)) {
108 bs.clear(m->pos + static_cast<unsigned int>(n-m->min));
109 return true;
110 }
111 if (l == r) return false;
112 if (n < m->min)
113 r=m-1;
114 else
115 l=m+1;
116 }
117 GECODE_NEVER;
118 return false;
119 }
