174 Problem(const int n, const int* d);

178 int d(int i, int j) const;
180 const int* d(void) const;

185 inline
186 Problem::Problem(const int n, const int* d)
187 : _n(n), _d(d) {}
188 inline int
189 Problem::size(void) const {
190 return _n;
191 }
192 inline int
193 Problem::d(int i, int j) const {
194 return _d[i*_n+j];
195 }
196 inline const int*
197 Problem::d(void) const {
198 return _d;
199 }
200 inline int
201 Problem::max(void) const {
202 int m=0;
203 for (int i=0; i<_n*_n; i++)
204 m = std::max(m,_d[i]);
205 return m*_n;
206 }

214 const unsigned int ps_n = sizeof(ps) / sizeof(Problem);

227class TSP : public IntMinimizeScript {
228protected:
230 Problem p;
232 IntVarArray succ;
234 IntVar total;
235public:
237 TSP(const SizeOptions& opt)
238 : IntMinimizeScript(opt),
239 p(ps[opt.size()]),
240 succ(*this, p.size(), 0, p.size()-1),
241 total(*this, 0, p.max()) {
242 int n = p.size();
243
244 // Cost matrix
245 IntArgs c(n*n, p.d());
246
247 // Cost of each edge
248 IntVarArgs costs(*this, n, Int::Limits::min, Int::Limits::max);
249
250 // Enforce that the successors yield a tour with appropriate costs
251 circuit(*this, c, succ, costs, total, opt.ipl());
252
253 // Just assume that the circle starts forwards
254 {
255 IntVar p0(*this, 0, n-1);
256 element(*this, succ, p0, 0);
257 rel(*this, p0, IRT_LE, succ[0]);
258 }
259
260 // First enumerate cost values, prefer those that maximize cost reduction
261 branch(*this, costs, INT_VAR_REGRET_MAX_MAX(), INT_VAL_MIN());
262
263 // Then fix the remaining successors
264 branch(*this, succ, INT_VAR_MIN_MIN(), INT_VAL_MIN());
265 }
266
267 virtual IntVar cost(void) const {
268 return total;
269 }
270
271 TSP(TSP& s) : IntMinimizeScript(s), p(s.p) {
272 succ.update(*this, s.succ);
273 total.update(*this, s.total);
274 }
275
276 virtual Space*
277 copy(void) {
278 return new TSP(*this);
279 }
280
281 virtual void
282 print(std::ostream& os) const {
283 bool assigned = true;
284 for (int i=0; i<succ.size(); i++) {
285 if (!succ[i].assigned()) {
286 assigned = false;
287 break;
288 }
289 }
290 if (assigned) {
291 os << "\tTour: ";
292 int i=0;
293 do {
294 os << i << " -> ";
295 i=succ[i].val();
296 } while (i != 0);
297 os << 0 << std::endl;
298 os << "\tCost: " << total << std::endl;
299 } else {
300 os << "\tTour: " << std::endl;
301 for (int i=0; i<succ.size(); i++) {
302 os << "\t" << i << " -> " << succ[i] << std::endl;
303 }
304 os << "\tCost: " << total << std::endl;
305 }
306 }
307};
