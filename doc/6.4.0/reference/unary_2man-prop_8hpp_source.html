67
68 if (PL::advanced) {
69 GECODE_ES_CHECK(detectable(home,t));
70 GECODE_ES_CHECK(notfirstnotlast(home,t));
71 GECODE_ES_CHECK(edgefinding(home,t));
72 }
73
74 if (!PL::basic)
75 GECODE_ES_CHECK(subsumed(home,*this,t));
76
77 return ES_NOFIX;
