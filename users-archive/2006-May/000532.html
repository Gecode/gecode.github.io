   GC_Neighbored::propagate(Gecode::Space* home) {
     if (x0.assigned()) {
       GECODE_ME_CHECK(removeNonNeighbors(home,x1,x0.val()));
     }else if (x1.assigned()) {
       GECODE_ME_CHECK(removeNonNeighbors(home,x0,x1.val()));
     } else {
       GECODE_ME_CHECK(removeNonNeighbors(home));
