class MyConstraint : public Space
{

              post(this, x[0] == 0);
              post(this, y[0] == 1);
              status();

        MyConstraint(bool share, MyConstraint& cp) : Space(share, cp)
        {
            x.update(this, share, cp.x);
            y.update(this, share, cp.y);

        {
            return new MyConstraint(share, *this);
        }
