var float: F;
var float: R;
var float: N;
var float: I;
var float: L;
var float: k;

constraint F=(R*R*N*N*I*I*k)/(2.0*L);

constraint F=20.0;

solve satisfy;

output [
        "F   : ", show(F),    "\n",
        "R   : ", show(R),    "\n",
        "N   : ", show(N),    "\n",
        "I   : ", show(I),    "\n",
        "L   : ", show(L),    "\n",
        "k   : ", show(k),    "\n"
];
